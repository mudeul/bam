const pages = dv.pages('""')
    .where(p => p.file.folder && p.file.folder.includes("01. 꿈") && !p.file.name.includes("전체보기"))
    .sort(p => p.file.ctime, 'desc');

for (let page of pages) {
    // 노트 생성일 포맷팅
    const date = moment(page.file.ctime.toString()).format("YYYY-MM-DD");
    
    // 노트 파일 내용 읽어오기
    const content = await dv.io.load(page.file.path);
    
    // 본문에서 첫 번째 의미 있는 문단 추출 (제목/속성 제외)
    const lines = content.split("\n").filter(line => 
        line.trim() !== "" && 
        !line.startsWith("---") && 
        !line.startsWith("#") &&
        !line.includes("cssclasses:")
    );
    const preview = lines.length > 0 ? lines[0] : "*(내용 없음)*";

    // 타임라인 형태 출력
    dv.paragraph(`#### ${page.file.link} <span style="font-size: 0.8em; color: gray; font-weight: normal;">| ${date}</span>\n> ${preview}\n---`);
}
